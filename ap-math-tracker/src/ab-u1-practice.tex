\partheading{Part A \textbullet\ Multiple choice \textbullet\ No calculator}
\begin{directions}[Directions \textbullet\ Part A]
A calculator may \textbf{not} be used. Solve each problem, then choose the best of the four answer choices.
\end{directions}

\begin{exo}[1-2]{}
$\displaystyle\lim_{x\to3}\frac{x^2-9}{x^2-2x-3}$ is
\choices{$0$}{$\dfrac32$}{$3$}{nonexistent}
\answer{B}{$\dfrac{(x-3)(x+3)}{(x-3)(x+1)}=\dfrac{x+3}{x+1}\to\dfrac64=\dfrac32$.}
\end{exo}
\begin{exo}[1-2]{}
If $f(x)=\begin{cases}x^2+1, & x<2\\ 3x-1, & x\ge2,\end{cases}$ then $\displaystyle\lim_{x\to2}f(x)$ is
\choices{$5$}{$4$}{$6$}{nonexistent}
\answer{A}{Left: $5$. Right: $6-1=5$. Equal, so the limit is $5$.}
\end{exo}
\begin{exo}[1-3]{}
$\displaystyle\lim_{x\to\infty}\frac{4x^3-x}{2x^3+7x^2}$ is
\choices{$0$}{$\dfrac47$}{$2$}{nonexistent}
\answer{C}{Same degree: ratio of the leading coefficients, $\frac42=2$.}
\end{exo}
\begin{exo}[1-3]{}
Which line or lines are vertical asymptotes of the graph of $h(x)=\dfrac{x+1}{x^2-1}$?
\choices{$x=1$ and $x=-1$}{$x=-1$ only}{$y=0$ only}{$x=1$ only}
\answer{D}{$\frac{x+1}{(x-1)(x+1)}$: $x+1$ cancels (hole at $x=-1$); $x-1$ stays (asymptote $x=1$).}
\end{exo}
\begin{exo}[1-3]{}
$\displaystyle\lim_{x\to2^-}\frac{x+1}{x-2}$ is
\choices{$-\infty$}{$+\infty$}{$-\dfrac32$}{$0$}
\answer{A}{Numerator near $3$, denominator small and negative: $-\infty$.}
\end{exo}
\begin{exo}[1-4]{}
For what value of $k$ is $f(x)=\begin{cases}kx+3, & x\le2\\ x^2-k, & x>2\end{cases}$ continuous at $x=2$?
\choices{$1$}{$\dfrac13$}{$-\dfrac13$}{$\dfrac73$}
\answer{B}{$2k+3=4-k$, so $3k=1$ and $k=\frac13$.}
\end{exo}
\begin{exo}[1-4]{}
What type of discontinuity does $g(x)=\dfrac{x^2-4}{x-2}$ have at $x=2$?
\choices{Jump}{Infinite}{Removable}{None}
\answer{C}{$\lim_{x\to2}g(x)=\lim(x+2)=4$ exists, but $g(2)$ is undefined: a hole.}
\end{exo}
\begin{exo}[1-5]{}
The function $f$ is continuous on $[1,5]$ with $f(1)=-2$ and $f(5)=6$. Which statement must be true?
\choices{$f(3)=2$}{$f$ is increasing on $[1,5]$.}{The maximum of $f$ on $[1,5]$ is $6$.}{$f(c)=0$ for some $c$ in $(1,5)$.}
\answer{D}{IVT: $0$ is between $-2$ and $6$.}
\end{exo}
\begin{exo}[1-5]{}
Which function is guaranteed by the Extreme Value Theorem to have an absolute maximum on the given interval?
\choices{$f(x)=x^2-3x$ on $[0,4]$}{$f(x)=\dfrac1x$ on $(0,1]$}{$f(x)=x$ on $(0,2)$}{$f(x)=\dfrac{1}{x-1}$ on $[0,2]$}
\answer{A}{Continuous on a closed interval.}
\end{exo}
\begin{exo}[1-6]{}
What is the average rate of change of $f(x)=x^3-2x$ on $[-1,2]$?
\choices{$3$}{$1$}{$\dfrac53$}{$-1$}
\answer{B}{$f(2)=4$, $f(-1)=1$: $\frac{4-1}{3}=1$.}
\end{exo}
\begin{exo}[1-6]{}
$V(t)$ is the volume of water in a tank, in liters, $t$ minutes after noon, with $V(2)=50$ and $V(8)=20$. Which is the average rate of change of $V$ on $[2,8]$?
\choices{$-30$ liters}{$5$ liters per minute}{$-5$ liters per minute}{$-\dfrac15$ minutes per liter}
\answer{C}{$\frac{20-50}{8-2}=-5$ liters per minute.}
\end{exo}
\begin{exo}[1-7]{}
At which point is $f(x)=|x-2|+1$ continuous but not differentiable?
\choices{$x=0$}{$x=1$}{$x=-2$}{$x=2$}
\answer{D}{Sharp corner at $x=2$.}
\end{exo}
\begin{exo}[1-8]{}
The function $f$ is differentiable with $f(1)=4$ and $f(5)=-4$. Which statement must be true?
\choices{$f'(c)=-2$ for some $c$ in $(1,5)$}{$f'(c)=2$ for some $c$ in $(1,5)$}{$f'(c)=-8$ for some $c$ in $(1,5)$}{$f(3)=0$}
\answer{A}{MVT: $\frac{-4-4}{5-1}=-2$.}
\end{exo}
\begin{exo}[1-9]{}
The depth of a river, $D(t)$, is differentiable. A question asks: ``Was the depth ever changing at exactly the average rate for the day?'' Which theorem answers it?
\choices{Intermediate Value Theorem}{Mean Value Theorem}{Extreme Value Theorem}{None of them}
\answer{B}{A rate of change (slope) equal to an average rate: MVT.}
\end{exo}

\begin{exo}[1-1]{}
Which of the following limits does \textbf{not} exist?
\choices{$\displaystyle\lim_{x\to1}g(x)$}{$\displaystyle\lim_{x\to-1^+}g(x)$}{$\displaystyle\lim_{x\to-1}g(x)$}{$\displaystyle\lim_{x\to4}g(x)$}
\answer{C}{At $x=-1$ the left-hand limit is $2$ and the right-hand limit is $0$.}
\end{exo}
\begin{exo}[1-1]{}
What is $\displaystyle\lim_{x\to1}g(x)$?
\choices{$4$}{$2$}{$3$}{nonexistent}
\answer{B}{Both pieces approach the open circle $(1,2)$; $g(1)=4$ does not change the limit.}
\end{exo}
\begin{exo}[1-3]{}
Which statement about the behavior of $g$ near $x=3$ is true?
\choices{$\displaystyle\lim_{x\to3^-}g(x)=+\infty$ and $\displaystyle\lim_{x\to3^+}g(x)=-\infty$}{$\displaystyle\lim_{x\to3}g(x)=+\infty$}{$\displaystyle\lim_{x\to3}g(x)=2$}{$g(3)=2$}
\answer{A}{The graph rises without bound on the left of $x=3$ and falls without bound on the right.}
\end{exo}
\begin{exo}[1-4]{}
At which value of $x$ does $g$ have a removable discontinuity?
\choices{$x=-1$}{$x=3$}{$x=-4$}{$x=1$}
\answer{D}{At $x=1$ the limit is $2$ but $g(1)=4$.}
\end{exo}
\begin{exo}[1-6]{}
What is the average rate of change of $g$ on $[-4,0]$?
\choices{$\dfrac14$}{$-1$}{$-\dfrac14$}{$\dfrac34$}
\answer{C}{$g(-4)=2$, $g(0)=1$: $\frac{1-2}{4}=-\frac14$.}
\end{exo}
\begin{exo}[1-7]{}
At which value of $x$ in $(-4,3)$ is $g$ differentiable?
\choices{$x=0$}{$x=-1$}{$x=1$}{$x=3$}
\answer{A}{At $x=0$ the graph is a straight line (slope $1$). $g$ is not continuous at $x=-1$ (jump), $x=1$ (hole) or $x=3$ (asymptote), so it is not differentiable there.}
\end{exo}

\begin{exo}[Graph analysis]{}
The graph of the function $f$ on $[-3,4]$ is shown. It consists of a line segment for $-3\le x<-1$, the parabola $y=x^2$ for $-1\le x<2$, and a line segment for $2<x\le4$. Also $f(-1)=1$ and $f(2)=1$.
\begin{center}
\begin{tikzpicture}
\begin{axis}[apaxis,xmin=-3.5,xmax=4.5,ymin=-1,ymax=5,xtick={-3,-2,-1,1,2,3,4},ytick={1,2,3,4},width=8.5cm,height=5.8cm]
\addplot[apblue,very thick,domain=-3:-1]{x+4};
\addplot[apblue,very thick,domain=-1:2]{x^2};
\addplot[apblue,very thick,domain=2:4]{8-2*x};
\node[circle,draw=apblue,fill=white,thick,inner sep=1.6pt] at (axis cs:-1,3){};
\node[circle,fill=apblue,inner sep=1.6pt] at (axis cs:-1,1){};
\node[circle,draw=apblue,fill=white,thick,inner sep=1.6pt] at (axis cs:2,4){};
\node[circle,fill=apblue,inner sep=1.6pt] at (axis cs:2,1){};
\end{axis}
\end{tikzpicture}
\end{center}
\begin{enumerate}[label=(\alph*)]
\item Find $\displaystyle\lim_{x\to-1^-}f(x)$ and $\displaystyle\lim_{x\to-1^+}f(x)$. Does $\displaystyle\lim_{x\to-1}f(x)$ exist? Give a reason for your answer.
\item Find $\displaystyle\lim_{x\to2}f(x)$. Is $f$ continuous at $x=2$? Name the type of discontinuity, if any, and explain how $f(2)$ could be redefined to make $f$ continuous at $x=2$.
\item Is $f$ differentiable at $x=-1$? At $x=0$? Explain each answer.
\item $f(1)=1$ and $f(3)=2$. Explain why the Intermediate Value Theorem cannot be used to guarantee a value $c$ in $[1,3]$ with $f(c)=1.5$. Does such a value exist? Justify.
\end{enumerate}
\begin{solution}
\textbf{(a)} $\lim_{x\to-1^-}f(x)=3$ and $\lim_{x\to-1^+}f(x)=(-1)^2=1$. The one-sided limits are not equal, so $\lim_{x\to-1}f(x)$ does not exist.\\
\textbf{(b)} $\lim_{x\to2^-}f(x)=2^2=4$ and $\lim_{x\to2^+}f(x)=8-2(2)=4$, so $\lim_{x\to2}f(x)=4$. But $f(2)=1\neq4$, so $f$ is not continuous at $x=2$. This is a removable discontinuity; redefining $f(2)=4$ makes $f$ continuous at $x=2$.\\
\textbf{(c)} At $x=-1$: no, $f$ is not continuous there (jump), and a function is not differentiable where it is not continuous. At $x=0$: yes, the graph is the smooth parabola $y=x^2$ there, with a horizontal tangent line (slope $0$).\\
\textbf{(d)} The IVT requires $f$ to be continuous on $[1,3]$, but $f$ is not continuous at $x=2$. Such a value does exist: on $[1,2)$, $f(x)=x^2=1.5$ when $c=\sqrt{1.5}\approx1.22$, which is in $[1,3]$.
\end{solution}
\end{exo}

\begin{exo}[Algebraic limits and continuity]{}
Let $g(x)=\begin{cases}\dfrac{x^2+x-6}{x-2}, & x<2\\[6pt] cx+1, & x\ge2\end{cases}$ \ where $c$ is a constant, and let $h(x)=\dfrac{x^2+x-6}{x^2-4}$.
\begin{enumerate}[label=(\alph*)]
\item Find $\displaystyle\lim_{x\to2^-}g(x)$. Show the work that leads to your answer.
\item Find the value of $c$ for which $g$ is continuous at $x=2$. Justify your answer using the definition of continuity.
\item Find all values of $x$ where $h$ is not continuous, and classify each discontinuity. Justify with limits.
\item Find $\displaystyle\lim_{x\to\infty}h(x)$ and $\displaystyle\lim_{x\to-2^+}h(x)$.
\end{enumerate}
\begin{solution}
\textbf{(a)} For $x<2$: $\dfrac{(x+3)(x-2)}{x-2}=x+3$, so $\lim_{x\to2^-}g(x)=5$.\\
\textbf{(b)} $g(2)=2c+1$ and $\lim_{x\to2^+}g(x)=2c+1$. For continuity we need $\lim_{x\to2^-}g(x)=\lim_{x\to2^+}g(x)=g(2)$: $2c+1=5$, so $c=2$. Then $g(2)=5$ and $\lim_{x\to2}g(x)=5=g(2)$.\\
\textbf{(c)} $h(x)=\dfrac{(x+3)(x-2)}{(x+2)(x-2)}$ is not defined at $x=2$ and $x=-2$.
At $x=2$: $\lim_{x\to2}h(x)=\lim_{x\to2}\dfrac{x+3}{x+2}=\dfrac54$ exists: removable discontinuity.
At $x=-2$: the numerator of $\dfrac{x+3}{x+2}$ approaches $1$ and the denominator approaches $0$, so $\lim_{x\to-2^-}h(x)=-\infty$ and $\lim_{x\to-2^+}h(x)=+\infty$: infinite discontinuity (vertical asymptote $x=-2$).\\
\textbf{(d)} $\lim_{x\to\infty}h(x)=1$ (equal degrees, leading coefficients $1$ and $1$). $\lim_{x\to-2^+}h(x)=+\infty$.
\end{solution}
\end{exo}

\begin{exo}[Table and theorems]{}
The temperature $W$, in degrees Celsius, of a greenhouse is a differentiable function of time $t$, in hours, for $0\le t\le10$. Selected values are given in the table.
\begin{center}\begin{tabular}{|c|c|c|c|c|c|}\hline
$t$ (hours) & 0 & 3 & 5 & 8 & 10\\\hline $W(t)$ ($^\circ$C) & 20 & 24 & 31 & 27 & 22\\\hline\end{tabular}\end{center}
\begin{enumerate}[label=(\alph*)]
\item Must there be a time $t$ in $[0,10]$ at which $W(t)=30$? Justify your answer.
\item Explain why there must be at least two times in $[0,10]$ at which $W(t)=25$.
\item Find the average rate of change of $W$ on $3\le t\le8$. Give units and interpret it.
\item Must there be a time $t$ in $(5,8)$ at which $W'(t)=-\frac43$? Justify. Must $W$ have an absolute maximum on $[0,10]$? Justify.
\end{enumerate}
\begin{solution}
\textbf{(a)} Yes. $W$ is differentiable, so it is continuous on $[3,5]$; $W(3)=24$ and $W(5)=31$. Since $24<30<31$, the IVT guarantees a $t$ in $(3,5)$ with $W(t)=30$.\\
\textbf{(b)} On $[3,5]$: $24<25<31$, so $W(t)=25$ for some $t$ in $(3,5)$. On $[8,10]$: $22<25<27$, so $W(t)=25$ for some $t$ in $(8,10)$. These intervals do not overlap: at least two times (IVT twice).\\
\textbf{(c)} $\dfrac{W(8)-W(3)}{8-3}=\dfrac{27-24}{5}=0.6$ $^\circ$C per hour: between hours $3$ and $8$ the temperature rose on average $0.6$ degrees per hour.\\
\textbf{(d)} Yes: $W$ is differentiable on $(5,8)$ and continuous on $[5,8]$, and $\dfrac{27-31}{8-5}=-\dfrac43$, so the MVT applies. Yes: $W$ is continuous on the closed interval $[0,10]$, so the EVT guarantees an absolute maximum (at least $31$).
\end{solution}
\end{exo}
