%%DOC 2.1A | Equivalent Expressions
\begin{exo}{}
Which expression is equivalent to $(2x+3)(x-4)$?
\choices{$2x^2-5x-12$}{$2x^2-11x-12$}{$2x^2+5x-12$}{$2x^2-12$}
\answer{A}{$2x^2-8x+3x-12=2x^2-5x-12$.}
\end{exo}
\begin{exo}{}
Which expression is equivalent to $9x^2-25$?
\choices{$(3x-5)^2$}{$(9x-5)(x+5)$}{$(3x-5)(3x+5)$}{$(3x-25)(3x+1)$}
\answer{C}{Difference of squares: $(3x)^2-5^2=(3x-5)(3x+5)$.}
\end{exo}
\begin{exo}{}
Which expression is equivalent to $(3x^2-2x+5)-(x^2+4x-1)$?
\choices{$2x^2+2x+4$}{$2x^2-6x+6$}{$2x^2-6x+4$}{$4x^2+2x+4$}
\answer{B}{Distribute the minus: $3x^2-2x+5-x^2-4x+1=2x^2-6x+6$.}
\end{exo}
\begin{exo}{}
For $x>0$, which expression is equivalent to $x^{3/4}$?
\choices{$\sqrt[3]{x^4}$}{$\dfrac{x^3}{4}$}{$\dfrac{3x}{4}$}{$\sqrt[4]{x^3}$}
\answer{D}{$x^{m/n}=\sqrt[n]{x^m}$: the denominator $4$ is the index.}
\end{exo}
\begin{exo}{\sprtag}
If $x^2+bx+16=(x+4)^2$ for all values of $x$, what is the value of $b$?
\spr{8}{8}{$(x+4)^2=x^2+8x+16$.}
\end{exo}
\begin{exo}{}
For $x\neq-2$ and $x\neq-3$, which expression is equivalent to $\dfrac{x^2-9}{x^2+5x+6}$?
\choices{$\dfrac{x-3}{x+2}$}{$\dfrac{x+3}{x+2}$}{$\dfrac{-9}{5x+6}$}{$\dfrac{x-3}{x-2}$}
\answer{A}{$\dfrac{(x-3)(x+3)}{(x+2)(x+3)}=\dfrac{x-3}{x+2}$.}
\end{exo}
\begin{exo}{}
Which expression is equivalent to $(2x^3y)(3x^2y^4)$?
\choices{$5x^5y^5$}{$6x^6y^4$}{$6x^5y^5$}{$6x^5y^4$}
\answer{C}{Multiply the coefficients and add the exponents: $6x^{3+2}y^{1+4}$.}
\end{exo}
\begin{exo}{\sprtag}
In the equation $(ax+3)(2x-5)=6x^2+bx-15$, $a$ and $b$ are constants, and the equation is true for all $x$. What is the value of $b$?
\spr{$-9$}{-9}{Expanding gives $2ax^2+(6-5a)x-15$. So $2a=6$, $a=3$, and $b=6-15=-9$.}
\end{exo}
\begin{exo}{}
For $x\neq1$ and $x\neq-1$, which expression is equivalent to $\dfrac{1}{x-1}+\dfrac{2}{x+1}$?
\choices{$\dfrac{3}{2x}$}{$\dfrac{3x-1}{x^2-1}$}{$\dfrac{3x+1}{x^2-1}$}{$\dfrac{3}{x^2-1}$}
\answer{B}{$\dfrac{(x+1)+2(x-1)}{(x-1)(x+1)}=\dfrac{3x-1}{x^2-1}$.}
\end{exo}
\begin{exo}{\sprtag}
The expression $4x^2-12x+9$ can be written as $(ax-b)^2$, where $a$ and $b$ are positive constants. What is the value of $a+b$?
\spr{5}{5}{$(2x-3)^2=4x^2-12x+9$, so $a=2$, $b=3$ and $a+b=5$.}
\end{exo}

%%DOC 2.2A | Nonlinear Equations and Systems
\begin{exo}{}
What are the solutions of $x^2-5x-14=0$?
\choices{$-7$ and $2$}{$7$ and $-2$}{$14$ and $-1$}{$5$ and $-14$}
\answer{B}{$(x-7)(x+2)=0$.}
\end{exo}
\begin{exo}{}
Which value of $x$ is a solution of $(x-3)^2=25$?
\choices{$2$}{$28$}{$-8$}{$8$}
\answer{D}{$x-3=\pm5$, so $x=8$ or $x=-2$.}
\end{exo}
\begin{exo}{}
How many distinct real solutions does $2x^2-3x+5=0$ have?
\choices{Zero}{Exactly one}{Exactly two}{Infinitely many}
\answer{A}{The discriminant is $(-3)^2-4(2)(5)=9-40=-31<0$.}
\end{exo}
\begin{exo}{\sprtag}
What is the solution of $\sqrt{2x+3}=x$?
\spr{3}{3}{$2x+3=x^2 \Rightarrow (x-3)(x+1)=0$. Check: $x=3$ gives $\sqrt9=3$; $x=-1$ gives $\sqrt1=1\neq-1$, so it is extraneous.}
\end{exo}
\begin{exo}{}
What is the sum of the solutions of $|x-4|=6$?
\choices{$4$}{$8$}{$12$}{$-8$}
\answer{B}{$x-4=6$ or $x-4=-6$: $x=10$ or $x=-2$, sum $8$.}
\end{exo}
\begin{exo}{}
What are the solutions of $x^2-4x-1=0$?
\choices{$-2\pm\sqrt5$}{$4\pm\sqrt5$}{$2\pm\sqrt5$}{$2\pm\sqrt3$}
\answer{C}{$x=\dfrac{4\pm\sqrt{16+4}}{2}=\dfrac{4\pm2\sqrt5}{2}=2\pm\sqrt5$.}
\end{exo}

\begin{exo}{\sprtag}
The graphs of $y=x^2$ and $y=4x-4$ are shown. The system of these two equations has exactly one solution $(x,y)$. What is the value of $y$?
\gr{xmin=-1.5,xmax=4.5,ymin=-1.5,ymax=9.5,width=5.5cm,height=5.6cm}{\addplot[apblue,very thick,domain=-1.5:3.1,samples=60]{x^2};\addplot[apred,very thick,domain=0.5:3.3]{4*x-4};}
\spr{4}{4}{$x^2=4x-4 \Rightarrow (x-2)^2=0$, so $x=2$ and $y=4$.}
\end{exo}
\begin{exo}{}
What is the solution of $\dfrac{5}{x-2}=\dfrac{3}{x}$?
\choices{$3$}{$-1$}{$-3$}{$6$}
\answer{C}{Cross-multiply: $5x=3x-6$, so $x=-3$. Neither denominator is zero.}
\end{exo}
\begin{exo}{\sprtag}
If $2^{x+1}=32$, what is the value of $x$?
\spr{4}{4}{$32=2^5$, so $x+1=5$.}
\end{exo}
\begin{exo}{}
For which value of $k$ does $x^2+kx+9=0$ have exactly one real solution?
\choices{$3$}{$9$}{$18$}{$-6$}
\answer{D}{One solution when $k^2-4(9)=0$: $k=6$ or $k=-6$. Only $-6$ is listed.}
\end{exo}

%%DOC 2.3A | Nonlinear Functions
\begin{exo}{}
What is the vertex of the graph of $f(x)=x^2-6x+5$?
\choices{$(-3,-4)$}{$(3,5)$}{$(3,-4)$}{$(6,5)$}
\answer{C}{$x=-\frac{-6}{2}=3$ and $f(3)=9-18+5=-4$.}
\end{exo}
\Needspace{18\baselineskip}
\begin{exo}{}
Which equation defines the parabola shown?
\gr{xmin=-2.5,xmax=4.5,ymin=-5,ymax=6,width=5.5cm,height=5.4cm}{\addplot[apblue,very thick,domain=-2:4,samples=60]{x^2-2*x-3};\pt{-1,0}\pt{3,0}\pt{1,-4}}
\choices{$y=(x+1)(x-3)$}{$y=(x-1)(x+3)$}{$y=(x+1)(x+3)$}{$y=(x-1)^2+4$}
\answer{A}{The $x$-intercepts are $-1$ and $3$ and the vertex $(1,-4)$ fits: $(2)(-2)=-4$.}
\end{exo}
\begin{exo}{}
A population of 500 bacteria triples every 4 hours. Which function gives the population $t$ hours later?
\choices{$P(t)=500\cdot3^{4t}$}{$P(t)=500\cdot3^{t/4}$}{$P(t)=1500^{t/4}$}{$P(t)=500+\frac{3t}{4}$}
\answer{B}{It is multiplied by $3$ each time $t$ increases by $4$.}
\end{exo}
\begin{exo}{\sprtag}
The function $f$ is defined by $f(x)=2^x+3$. What is the value of $f(5)$?
\spr{35}{35}{$2^5+3=32+3=35$.}
\end{exo}
\begin{exo}{}
A car worth $\$20{,}000$ loses $15\%$ of its value each year. Which function gives its value $t$ years later?
\choices{$V(t)=20000(0.15)^t$}{$V(t)=20000(1.15)^t$}{$V(t)=20000(0.85)^t$}{$V(t)=20000-0.15t$}
\answer{C}{Each year it keeps $85\%$ of its value: growth factor $1-0.15=0.85$.}
\end{exo}
\begin{exo}{\sprtag}
What is the minimum value of $g(x)=(x-2)^2+7$?
\spr{7}{7}{$(x-2)^2\ge0$, so the minimum is $7$, at $x=2$.}
\end{exo}
\begin{exo}{}
If $f(x)=x^2+1$ and $g(x)=2x-3$, what is the value of $f(g(2))$?
\choices{$5$}{$7$}{$1$}{$2$}
\answer{D}{$g(2)=1$, then $f(1)=2$.}
\end{exo}
\begin{exo}{}
A ball's height, in feet, $t$ seconds after it is thrown is $h(t)=-16t^2+64t+5$. What is its maximum height?
\choices{$64$}{$69$}{$5$}{$133$}
\answer{B}{The vertex is at $t=\frac{-64}{-32}=2$; $h(2)=-64+128+5=69$.}
\end{exo}
\begin{exo}{\sprtag}
The function $f$ is defined by $f(x)=a\cdot2^x$, where $a$ is a constant. If $f(3)=40$, what is the value of $a$?
\spr{5}{5}{$8a=40$, so $a=5$.}
\end{exo}
\begin{exo}{}
The graph of $y=f(x)$ is moved 3 units right and 2 units up. Which equation gives the new graph?
\choices{$y=f(x+3)+2$}{$y=f(x+2)-3$}{$y=f(x-3)-2$}{$y=f(x-3)+2$}
\answer{D}{A shift right by $3$ replaces $x$ with $x-3$; a shift up by $2$ adds $2$.}
\end{exo}
