\begin{exo}[Advanced Math]{}
For $x\neq4$, which expression is equivalent to $\dfrac{x^2-16}{x-4}$?
\choices{$x-4$}{$x+4$}{$x^2-4$}{$x+\dfrac{4}{x}$}
\answer{B}{$\frac{(x-4)(x+4)}{x-4}=x+4$.}
\end{exo}
\begin{exo}[Algebra]{}
In the system $kx+3y=8$ and $4x+6y=5$, $k$ is a constant. For what value of $k$ does the system have no solution?
\choices{$4$}{$-2$}{$8$}{$2$}
\answer{D}{Slopes $-\frac k3$ and $-\frac46$ are equal when $k=2$; then $2x+3y=8$ and $2x+3y=2.5$ are parallel.}
\end{exo}
\begin{exo}[Advanced Math]{\sprtag}

\end{exo}
\begin{exo}[Data Analysis]{}

\choices{The mean and the median both increase.}{The mean is unchanged and the median increases.}{The mean increases and the median is unchanged.}{Both are unchanged.}
\answer{C}{The sum grows by $18$, so the mean grows by $3.6$; the middle value is still $5$.}
\end{exo}
