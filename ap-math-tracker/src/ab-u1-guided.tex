\begin{nrow}{Limit notation}
\begin{tabular}{@{}p{0.31\linewidth}p{0.31\linewidth}p{0.3\linewidth}@{}}
\textbf{From the left} & \textbf{From the right} & \textbf{From both sides}\\[2pt]
\fillin{$\displaystyle\lim_{x\to c^-}f(x)$} & \fillin{$\displaystyle\lim_{x\to c^+}f(x)$} & \fillin{$\displaystyle\lim_{x\to c}f(x)$}\\[4pt]
\footnotesize $x$ approaches $c$ with values \fillin{less} than $c$ & \footnotesize $x$ approaches $c$ with values \fillin{greater} than $c$ & \footnotesize both one-sided limits are \fillin{equal}
\end{tabular}
\end{nrow}

\sg{xmin=-1.5,xmax=6.5,ymin=-0.5,ymax=6.5,width=8.5cm,height=5.6cm,xtick={-1,0,1,2,3,5,6},extra x ticks={4},extra x tick labels={$c$}}{\addplot[apblue,very thick,domain=-1:6,samples=80]{0.25*(x-2)^2+2};\draw[apred,dashed](axis cs:4,0)--(axis cs:4,3)--(axis cs:0,3);\pt{4,3}\node[apblue,font=\small] at (axis cs:5.6,4.8){$f(x)$};}
Use the graph, with $c=4$.\par
1) $\displaystyle\lim_{x\to4^-}f(x)=$ \fillin{$3$} \quad 2) $\displaystyle\lim_{x\to4^+}f(x)=$ \fillin{$3$} \quad 3) $\displaystyle\lim_{x\to4}f(x)=$ \fillin{$3$}\par
\smallskip A limit is a \fillin{$y$-value} (a height): the height the graph \fillin{approaches} as $x$ gets close to $c$.
\end{nrow}
\begin{nrow}{Challenge}
Draw the graph of a function whose left-hand and right-hand limits at $x=1$ are \textbf{not} the same.
\sg{xmin=-4.5,xmax=4.5,ymin=-3.5,ymax=4.5,width=7cm,height=5.2cm}{\ifkey\addplot[apblue,very thick,domain=-4:1]{x+2};\addplot[apblue,very thick,domain=1:4]{-x+1};\pt{1,3}\op{1,0}\fi}
\ifkey{\color{apnavy}\small Example: the left piece arrives at height $3$, the right piece starts at height $0$, so $\lim_{x\to1}f(x)$ does not exist.}\fi
\end{nrow}
\begin{nrow}{Example:\\ one-sided limits are not the same}
\sg{xmin=-3.5,xmax=5,ymin=-0.5,ymax=4.2,width=8.5cm,height=5.4cm}{\addplot[apblue,very thick,domain=-3:2]{0.5*x+2};\addplot[apblue,very thick,domain=2:4.5,samples=60]{(x-3)^2};\pt{2,3}\op{2,1}\draw[apgrey,dashed](axis cs:2,0)--(axis cs:2,4);\node[apblue,font=\small] at (axis cs:4.3,3.4){$g(x)$};}
1) $\displaystyle\lim_{x\to2^-}g(x)=$ \fillin{$3$} \qquad 2) $\displaystyle\lim_{x\to2^+}g(x)=$ \fillin{$1$}\par
3) $\displaystyle\lim_{x\to2}g(x)=$ \fillin{DNE} \qquad 4) $g(2)=$ \fillin{$3$} \;{\small(the height \emph{at} $x=2$: the filled dot)}\par
\smallskip If the left and right limits are different, the two-sided limit \fillin{does not exist} (DNE).
\end{nrow}
\begin{nrow}{Summary}
\ifkey{\color{apnavy}A limit is the height the graph approaches. $\lim_{x\to c}f(x)$ exists only when the left and right limits are equal. The value $f(c)$ is the height of the filled dot at $c$; it can be different from the limit, or not exist.}\else\vspace{2.6cm}\fi
\end{nrow}

\end{minipage}
\end{nrow}
\begin{nrow}{When plugging in fails}
$\displaystyle\lim_{x\to3}\frac{x^2-9}{x^2-x-6}$\quad Plug in: \fillin{$\dfrac{0}{0}$}, the \fillin{indeterminate form}. It means: do more work!\par\smallskip
\begin{minipage}[t]{0.3\linewidth}Three steps:\\ 1) \fillin{Factor}\\ 2) \fillin{Cancel}\\ 3) \fillin{Plug in}\end{minipage}\hfill
\begin{minipage}[t]{0.66\linewidth}
\begin{workspace}[2.4cm]
$\dfrac{(x-3)(x+3)}{(x-3)(x+2)}=\dfrac{x+3}{x+2}\;\to\;\dfrac{3+3}{3+2}=\dfrac65$
\end{workspace}
\end{minipage}
\end{nrow}
\begin{nrow}{Example 1}
$\displaystyle\lim_{x\to0}\frac{x^2-4x}{3x^2+x}$
\begin{workspace}[2.2cm]
$\frac00$. \;$\dfrac{x(x-4)}{x(3x+1)}=\dfrac{x-4}{3x+1}\to\dfrac{-4}{1}=-4$.
\end{workspace}
\end{nrow}
\begin{nrow}{Example 2}
$\displaystyle\lim_{x\to-3}\frac{x^2+6x+9}{x+3}$
\begin{workspace}[2.2cm]
$\frac00$. \;$\dfrac{(x+3)(x+3)}{x+3}=x+3\to-3+3=0$. \;(A limit of $0$ is a real answer, not DNE.)
\end{workspace}
\end{nrow}
\begin{nrow}{Summary}
\ifkey{\color{apnavy}Try plugging in first. A number means you are done; $\frac{0}{0}$ means factor, cancel, then plug in again. For a piecewise function, compare the left piece and the right piece at the suspect point.}\else\vspace{2.4cm}\fi
\end{nrow}

%%DOC 1-3 | Limits to Infinity and Horizontal Asymptotes
\begin{center}\textsf{\small\color{apgrey}Objective: (1) find vertical asymptotes and holes; (2) use the top/bottom rules for limits to infinity and horizontal asymptotes.}\end{center}
\begin{nrow}{The graph}
\sg{xmin=-5.5,xmax=6.5,ymin=-4.5,ymax=6.5,width=11cm,height=6.8cm}{\addplot[apblue,very thick,domain=-5.5:-1.18,samples=80]{(x+2)/(x+1)};\addplot[apblue,very thick,domain=-0.83:6.5,samples=100]{(x+2)/(x+1)};\va{-1}\ha{1}\op{2,1.3333}}
\centering Graph of $f(x)=\dfrac{x^2-4}{x^2-x-2}$
\end{nrow}
\begin{nrow}{Vertical asymptotes (VA)}
Factor: $f(x)=\dfrac{x^2-4}{x^2-x-2}=$ \fillin{$\dfrac{(x-2)(x+2)}{(x-2)(x+1)}$}\par\smallskip
\textbf{Vertical asymptote}: the \fillin{denominator} is $0$ after cancelling: \fillin{$x=-1$}.\par
\textbf{Hole}: a factor that \fillin{cancels}: \fillin{$x=2$}, at height $\frac{2+2}{2+1}=$ \fillin{$\frac43$}. A hole is a \fillin{removable} discontinuity.\par
Near a VA the limit is \fillin{$\pm\infty$}: \;$\displaystyle\lim_{x\to-1^+}f(x)=$ \fillin{$+\infty$}, \;$\displaystyle\lim_{x\to-1^-}f(x)=$ \fillin{$-\infty$}.
\end{nrow}
\begin{nrow}{Horizontal asymptotes (HA)}
A horizontal asymptote comes from a limit to \fillin{infinity}: $\displaystyle\lim_{x\to\infty}f(x)=$ \fillin{$1$}, so the HA is \fillin{$y=1$}.\par\smallskip
Fun fact: for a rational function with a HA, \fillin{$\displaystyle\lim_{x\to\infty}f(x)=\lim_{x\to-\infty}f(x)$}: the same HA on both ends.
\end{nrow}
\begin{nrow}{Limits to infinity: top/bottom rules}
Compare the \fillin{highest powers} (degrees) of the top and the bottom.\par\smallskip
1) $\displaystyle\lim_{x\to\pm\infty}\frac{\textsf{BIG}}{\textsf{small}}=$ \fillin{$\pm\infty$ (DNE, no HA)} \quad bigger degree on top\par
2) $\displaystyle\lim_{x\to\pm\infty}\frac{\textsf{small}}{\textsf{BIG}}=$ \fillin{$0$ (HA $y=0$)} \quad bigger degree on the bottom\par
3) $\displaystyle\lim_{x\to\pm\infty}\frac{\textsf{same}}{\textsf{same}}=$ \fillin{ratio of the leading coefficients} \quad same degree
\end{nrow}
\begin{nrow}{Examples}
1) $\displaystyle\lim_{x\to\infty}\frac{4x^3-x}{5-2x^2}=$ \fillin{DNE ($-\infty$)}\qquad
2) $\displaystyle\lim_{x\to-\infty}\frac{3x}{x^2+7}=$ \fillin{$0$}\par\smallskip
3) $\displaystyle\lim_{x\to\infty}\frac{6x^2+1}{3-2x^2}=$ \fillin{$\dfrac{6}{-2}=-3$}
\begin{workspace}[1.6cm]
1) degree $3>2$: grows without bound; $\frac{4x^3}{-2x^2}=-2x\to-\infty$. \;2) degree $1<2$: $0$. \;3) same degree $2$: $\frac{6}{-2}=-3$; HA $y=-3$.
\end{workspace}
\end{nrow}
\begin{nrow}{Summary}
\ifkey{\color{apnavy}Factor first: a cancelled factor gives a hole, a zero left in the denominator gives a vertical asymptote. For $x\to\pm\infty$, compare degrees: top bigger $\to\pm\infty$, bottom bigger $\to0$, equal $\to$ ratio of leading coefficients (the HA).}\else\vspace{2.4cm}\fi
\end{nrow}

\begin{nrow}{What does it mean to be continuous?}
A graph is continuous if you can trace it \fillin{without lifting your pencil}.\par\smallskip
Are these graphs continuous or discontinuous?\par
\begin{tabular}{@{}cccc@{}}

\end{tabular}
\end{nrow}
\begin{nrow}{Types of discontinuities}
\begin{tabular}{@{}ccc@{}}
\tinyg{xmin=-2.5,xmax=3.5,ymin=-1.5,ymax=4.5,width=4.6cm,height=3.9cm}{\addplot[apblue,very thick,domain=-2.4:3.4]{x+1};\op{1,2}\pt{1,-0.5}} &
\tinyg{xmin=-2.5,xmax=3.5,ymin=-1.5,ymax=4.5,width=4.6cm,height=3.9cm}{\addplot[apblue,very thick,domain=-2.4:1]{0.5*x+1};\addplot[apblue,very thick,domain=1:3.4]{0.5*x+2.5};\pt{1,1.5}\op{1,3}} &
\tinyg{xmin=-2.5,xmax=4,ymin=-1,ymax=5,width=4.6cm,height=3.9cm}{\addplot[apblue,very thick,domain=-2.4:0.55,samples=40]{1/(x-1)^2};\addplot[apblue,very thick,domain=1.45:3.9,samples=40]{1/(x-1)^2};\va{1}}\\
\fillin{Removable (hole)} & \fillin{Jump} & \fillin{Infinite (VA)}\\[2pt]
\footnotesize the limit \fillin{exists} but & \footnotesize the one-sided limits & \footnotesize the function goes\\
\footnotesize is not equal to $f(c)$ & \footnotesize are \fillin{different} & \footnotesize to \fillin{$\pm\infty$}
\end{tabular}\par\smallskip
An \textbf{endpoint} (like the start of $y=\sqrt x$ at $x=0$) only needs the one-sided limit from the \fillin{inside} of the domain.
\end{nrow}
\begin{nrow}{Continuity test}
$f$ is continuous at $x=c$ when
\[\fillin{$\displaystyle\lim_{x\to c^-}f(x)$}\;=\;\fillin{$\displaystyle\lim_{x\to c^+}f(x)$}\;=\;\fillin{$f(c)$}\]
\centering{\small\color{apgrey}\qquad left \hspace{2.4cm} right \hspace{2.2cm} actual height}
\end{nrow}
\begin{nrow}{Example 1}
$h(x)=\begin{cases}2x-1, & x\le3\\ x^2-4, & x>3\end{cases}$\qquad Prove that $h$ is continuous at $x=3$.
\begin{workspace}[2.6cm]
Left: $\lim_{x\to3^-}h(x)=2(3)-1=5$. \;Right: $\lim_{x\to3^+}h(x)=3^2-4=5$. \;Height: $h(3)=2(3)-1=5$.\\
$5=5=5$, so $h$ is continuous at $x=3$ because it passes the continuity test.
\end{workspace}
\end{nrow}
\begin{nrow}{Example 2}
For what value of $k$ is $g$ continuous at $x=1$? \qquad $g(x)=\begin{cases}3x^2-k, & x\le1\\ kx+5, & x>1\end{cases}$
\begin{workspace}[2.8cm]
Set left $=$ right at $x=1$: $3(1)^2-k=k(1)+5$, so $3-k=k+5$, $-2=2k$, $k=-1$.\\
Check: left $3-(-1)=4$, right $-1+5=4$, $g(1)=4$. Continuous.
\end{workspace}
\end{nrow}
\begin{nrow}{Summary}
\ifkey{\color{apnavy}Continuous means no hole, no jump and no asymptote at the point. To prove it, show left limit $=$ right limit $=f(c)$. To find a constant, set the two pieces equal at the suspect point and solve.}\else\vspace{2.4cm}\fi
\end{nrow}

%%DOC 1-5 | IVT and EVT
\begin{center}\textsf{\small\color{apgrey}Objective: use the Intermediate Value Theorem and the Extreme Value Theorem, and know what each one needs.}\end{center}
\begin{nrow}{Symbolic notation of graphs}
\sg{xmin=-2.5,xmax=6.5,ymin=-0.5,ymax=5,width=9cm,height=5.2cm,xtick={-1,2,4.5},xticklabels={$a$,$c$,$b$},ytick=\empty}{\addplot[apblue,very thick,domain=-2.3:6.3,samples=80]{-0.25*(x-2)^2+4};\pt{-1,1.75}\pt{2,4}\pt{4.5,2.4375}\ifkey\node[apnavy,font=\small,anchor=south east] at (axis cs:-1,1.75){$(a,f(a))$};\node[apnavy,font=\small,anchor=south] at (axis cs:2,4.1){$(c,f(c))$};\node[apnavy,font=\small,anchor=south west] at (axis cs:4.5,2.5){$(b,f(b))$};\fi}
A point on the graph is written \fillin{$(x,f(x))$}: $f(a)$ is the \fillin{height} of the graph at $x=a$.\par\smallskip
\begin{tabular}{@{}p{0.47\linewidth}p{0.47\linewidth}@{}}
\textbf{Maximum} at $x=c$: & \textbf{Minimum} at $x=c$:\\
\fillin{$f(c)\ge f(x)$} for every $x$ in the interval & \fillin{$f(c)\le f(x)$} for every $x$ in the interval\\
\footnotesize one height, at $x=c$, is above \fillin{all} the other heights & \footnotesize one height is below all the others
\end{tabular}
\end{nrow}
\begin{nrow}{Types of intervals}
\begin{tabular}{@{}cc@{}}
\tinyg{xmin=-0.5,xmax=4.5,ymin=-0.5,ymax=3.5,width=5.4cm,height=3.8cm}{\addplot[apblue,very thick,domain=0.5:4,samples=40]{1+sin(deg(x))*1.5};\pt{0.5,1.719}\pt{4,-0.135}} &
\tinyg{xmin=-0.5,xmax=4.5,ymin=-0.5,ymax=3.5,width=5.4cm,height=3.8cm}{\addplot[apblue,very thick,domain=0.5:4,samples=40]{1+sin(deg(x))*1.5};\op{0.5,1.719}\op{4,-0.135}}\\
\fillin{Closed} interval: endpoints included & \fillin{Open} interval: endpoints excluded\\
inequality \fillin{$a\le x\le b$}, \;interval \fillin{$[a,b]$} & inequality \fillin{$a<x<b$}, \;interval \fillin{$(a,b)$}
\end{tabular}
\end{nrow}
\begin{nrow}{Intermediate Value Theorem (IVT)}
\begin{tabular}{|l|p{0.74\linewidth}|}\hline
\textbf{Needs} & If $f$ is \fillin{continuous} on $[a,b]$,\\\hline
\textbf{Proves} & then $f$ takes every \fillin{height (value)} between $f(a)$ and $f(b)$ at least \fillin{once}.\\\hline
\end{tabular}\par\smallskip
\begin{minipage}[t]{0.47\linewidth}\vspace{0pt}
\tinyg{xmin=-0.5,xmax=5.5,ymin=-3,ymax=7,width=5.6cm,height=4.4cm}{\addplot[apblue,very thick,domain=1:5,samples=60]{-2+8*((x-1)/4)^0.6+sin(deg(3*x))};\pt{1,-1.86}\pt{5,6.65}\draw[apred,dashed](axis cs:-1,0)--(axis cs:6,0);}
\end{minipage}\hfill
\begin{minipage}[t]{0.5\linewidth}\vspace{2pt}
$f$ is continuous on $[1,5]$, $f(1)<0<f(5)$.\par The IVT guarantees a value $c$ in \fillin{$(1,5)$} with $f(c)=$ \fillin{$0$}: the graph cannot go from below the $x$-axis to above it without \fillin{crossing} it.
\end{minipage}
\par\smallskip
\textbf{Try it.} $g$ is continuous. \;\begin{tabular}{|c|c|c|c|c|}\hline$x$&$0$&$2$&$4$&$6$\\\hline$g(x)$&$3$&$-1$&$5$&$8$\\\hline\end{tabular}\; At least how many zeros does $g$ have on $[0,6]$?
\begin{workspace}[1.6cm]
At least $2$: $g$ changes sign on $[0,2]$ ($3$ to $-1$) and on $[2,4]$ ($-1$ to $5$). The IVT gives a zero in each interval.
\end{workspace}
\end{nrow}
\begin{nrow}{Extreme Value Theorem (EVT)}
\begin{tabular}{|l|p{0.74\linewidth}|}\hline
\textbf{Needs} & If $f$ is \fillin{continuous} on a \fillin{closed} interval $[a,b]$,\\\hline
\textbf{Proves} & then $f$ has an absolute \fillin{maximum} and an absolute \fillin{minimum} on $[a,b]$.\\\hline
\end{tabular}\par\smallskip
\begin{tabular}{@{}cc@{}}
\tinyg{xmin=-0.5,xmax=5.5,ymin=-0.5,ymax=4.5,width=5.4cm,height=3.8cm}{\addplot[apblue,very thick,domain=0.5:5,samples=60]{2+1.5*sin(deg(1.3*x))};\pt{0.5,2.9}\pt{5,2.32}\ifkey\node[apgreen,font=\scriptsize\bfseries] at (axis cs:1.2,4.1){max};\node[apred,font=\scriptsize\bfseries] at (axis cs:3.6,0.2){min};\fi} &
\tinyg{xmin=-0.5,xmax=5.5,ymin=-0.5,ymax=4.5,width=5.4cm,height=3.8cm}{\addplot[apblue,very thick,domain=0.5:4.5]{0.8*x};\op{0.5,0.4}\op{4.5,3.6}}\\
\footnotesize continuous on $[a,b]$: max and min \fillin{exist} & \footnotesize open interval: \fillin{no} max and \fillin{no} min
\end{tabular}\par
\footnotesize A horizontal segment on $[a,b]$ is a special case: every point is both a maximum and a minimum.
\end{nrow}
\begin{nrow}{Summary}
\ifkey{\color{apnavy}IVT needs continuity on $[a,b]$ and proves that every height between $f(a)$ and $f(b)$ is reached. EVT needs continuity on a closed interval and proves that an absolute maximum and minimum exist.}\else\vspace{2.4cm}\fi
\end{nrow}

\begin{center}\textsf{\small\color{apgrey}Objective: find and interpret the average rate of change on an interval, with units.}\end{center}
\begin{nrow}{Moussa's bike ride}
Moussa rides his bike along the Corniche in Dakar. The distance he has ridden, in meters, $t$ minutes after he starts, is $D(t)$ for $0\le t\le40$.
\begin{center}\begin{tabular}{|c|c|c|c|c|c|c|}\hline $t$ (min)&$0$&$5$&$10$&$20$&$30$&$40$\\\hline $D(t)$ (m)&$0$&$900$&$2000$&$4200$&$6900$&$9200$\\\hline\end{tabular}\end{center}
\sg{xmin=-1,xmax=42,ymin=-300,ymax=9800,width=10cm,height=5.4cm,xtick={0,5,10,20,30,40},ytick={0,2000,4000,6000,8000},scaled y ticks=false,yticklabel style={font=\tiny}}{\addplot[only marks,mark=*,apblue] coordinates {(0,0)(5,900)(10,2000)(20,4200)(30,6900)(40,9200)};\ifkey\addplot[apred,thick] coordinates {(0,0)(40,9200)};\addplot[apgreen!70!black,thick] coordinates {(5,900)(20,4200)};\fi}
\end{nrow}
\begin{nrow}{Average rate of change}
The average rate of change of $f$ on $a\le x\le b$:\qquad \fillin{$\dfrac{f(b)-f(a)}{b-a}$} $=\dfrac{\Delta y}{\Delta x}$\par\smallskip
Graphically, it is the \fillin{slope} of the \fillin{secant line} through the two points $(a,f(a))$ and $(b,f(b))$.\par
Units: \fillin{output units per input unit}.
\end{nrow}
\begin{nrow}{Example 1}
a) Sketch the line that represents the average rate of change of $D$ over the whole interval.\\
b) Find the average rate of change of $D$ on $0\le t\le40$. Give units.
\begin{workspace}[2cm]
a) The red line from $(0,0)$ to $(40,9200)$. \;b) $\dfrac{D(40)-D(0)}{40-0}=\dfrac{9200}{40}=230$ meters per minute.
\end{workspace}
\end{nrow}
\begin{nrow}{Example 2}
a) Sketch the line that represents the average rate of change of $D$ on $5\le t\le20$.\\
b) Find it and interpret it in context.
\begin{workspace}[2.2cm]
a) The green line from $(5,900)$ to $(20,4200)$. \;b) $\dfrac{4200-900}{20-5}=\dfrac{3300}{15}=220$ m/min: between minute $5$ and minute $20$, Moussa rode on average $220$ meters per minute.
\end{workspace}
\end{nrow}
\begin{nrow}{Example 3}
Find the average rate of change of $f(x)=x^2-2x$ on $[1,4]$.
\begin{workspace}[1.8cm]
$f(4)=8$, $f(1)=-1$: $\dfrac{8-(-1)}{4-1}=\dfrac93=3$.
\end{workspace}
\end{nrow}
\begin{nrow}{Summary}
\ifkey{\color{apnavy}Average rate of change $=\frac{f(b)-f(a)}{b-a}$, the slope of the secant line between two points. Always give units: output units per input unit (meters per minute).}\else\vspace{2.2cm}\fi
\end{nrow}

%%DOC 1-7 | Instantaneous Rates of Change
\begin{center}\textsf{\small\color{apgrey}Objective: draw tangent lines, read the sign of the derivative, and recognize where a graph is not differentiable.}\end{center}
\begin{nrow}{Rates on a graph}
\sg{xmin=-0.3,xmax=6.8,ymin=-0.5,ymax=5,width=9.5cm,height=5.2cm,xtick={0.5,2.057,3.5,6.2},xticklabels={$a$,$b$,$c$,$d$},ytick=\empty}{\addplot[apblue,very thick,domain=0:6.5,samples=120]{0.15*(x-1)*(x-3.5)*(x-6)+3};\pt{0.5,1.763}\pt{2.057,3.902}\pt{3.5,3}\pt{6.2,3.421}\ifkey\addplot[apred,thick,domain=0.1:0.9]{1.763+3.113*(x-0.5)};\addplot[apred,thick,domain=1.5:2.6]{3.902};\addplot[apred,thick,domain=2.9:4.1]{3-0.9375*(x-3.5)};\addplot[apred,thick,domain=5.85:6.55]{3.421+2.343*(x-6.2)};\node[apred,font=\scriptsize] at (axis cs:0.9,3.3){$+$};\node[apred,font=\scriptsize] at (axis cs:2.057,4.4){$0$};\node[apred,font=\scriptsize] at (axis cs:4.1,3.1){$-$};\node[apred,font=\scriptsize] at (axis cs:5.7,4.3){$+$};\fi}
The instantaneous rate of change at a point is the slope of the \fillin{tangent line}: the line that just touches the graph at that point.
\end{nrow}
\begin{nrow}{Synonyms}
Instantaneous rate of change $=$ \fillin{slope of the tangent line} $=$ \fillin{derivative} $=$ \fillin{$f'(c)$} $=$ \fillin{$\dfrac{dy}{dx}$}
\end{nrow}
\begin{nrow}{Example 1}
Sketch the tangent line at each point and decide whether the derivative is positive, negative or zero.\par
\begin{minipage}[t]{0.6\linewidth}\vspace{0pt}
\tinyg{xmin=-0.3,xmax=6.6,ymin=-0.5,ymax=5.5,width=7cm,height=4.6cm,xtick={0.5,1.5708,3,5.5},xticklabels={$a$,$b$,$c$,$d$}}{\addplot[apblue,very thick,domain=0:6.4,samples=100]{2.5+2*sin(deg(x))};\pt{0.5,3.459}\pt{1.5708,4.5}\pt{3,2.782}\pt{5.5,1.088}\ifkey\addplot[apred,thick,domain=0:1]{3.459+1.755*(x-0.5)};\addplot[apred,thick,domain=1:2.1]{4.5};\addplot[apred,thick,domain=2.5:3.5]{2.782-1.98*(x-3)};\addplot[apred,thick,domain=5:6]{1.088+1.418*(x-5.5)};\fi}
\end{minipage}\hfill
\begin{minipage}[t]{0.36\linewidth}\vspace{6pt}
$g'(a)$ \fillin{$>0$}\par\medskip $g'(b)$ \fillin{$=0$}\par\medskip $g'(c)$ \fillin{$<0$}\par\medskip $g'(d)$ \fillin{$>0$}
\end{minipage}
\end{nrow}
\begin{nrow}{Differentiability}
A function is \textbf{not differentiable} ($f'$ does not exist) at:\par
1) \fillin{discontinuities: holes, jumps, vertical asymptotes, endpoints}\par
2) \fillin{sharp corners (cusps)}\par
3) \fillin{vertical tangent lines}
\end{nrow}
\begin{nrow}{Example 2}
Find the $x$-coordinates where $h$ is not differentiable, and say why.
\sg{xmin=-5.8,xmax=10.5,ymin=-3.5,ymax=7,width=11cm,height=6cm,xtick={-5,-3,-1,1,3,5,7,9}}{\addplot[apblue,very thick] coordinates {(-5,1)(-3,-1)(1,3)};\pt{-5,1}\pt{1,3}\addplot[apblue,very thick,domain=1.02:6.75,samples=140]{2+sign(x-3)*abs(x-3)^(1/3)+0.6/(7-x)};\op{1,0.89}\op{5,3.56}\addplot[apblue,very thick,domain=7.25:10.3,samples=60]{3-1.5/(x-7)};\va{7}}
\begin{workspace}[2.4cm]
$x=-5$: endpoint. \;$x=-3$: sharp corner. \;$x=1$: jump (discontinuous). \;$x=3$: vertical tangent. \;$x=5$: hole (discontinuous). \;$x=7$: vertical asymptote.
\end{workspace}
\end{nrow}
\begin{nrow}{Summary}
\ifkey{\color{apnavy}The derivative $f'(c)$ is the slope of the tangent line at $x=c$: positive where the graph rises, negative where it falls, zero at a peak or valley. It does not exist at discontinuities, sharp corners and vertical tangents.}\else\vspace{2.2cm}\fi
\end{nrow}

\begin{center}\textsf{\small\color{apgrey}Objective: recognize when the Mean Value Theorem applies, and use it.}\end{center}
\begin{nrow}{Review: differentiability}
What is the instantaneous rate of change at each marked point: positive, negative, $0$ or DNE?\par
\begin{minipage}[t]{0.62\linewidth}\vspace{0pt}
\tinyg{xmin=-0.5,xmax=7.5,ymin=-0.5,ymax=4.5,width=7.4cm,height=4.6cm,xtick={1,2,3.5,4,7}}{\addplot[apblue,very thick,domain=0:4,samples=60]{3-(x-2)^2/2};\addplot[apblue,very thick] coordinates {(4,1)(7,4)};\pt{1,2.5}\pt{2,3}\pt{3.5,1.875}\pt{4,1}\pt{7,4}}
\end{minipage}\hfill
\begin{minipage}[t]{0.35\linewidth}\vspace{2pt}\small
$x=1$: \fillin{positive}\\ $x=2$: \fillin{$0$}\\ $x=3.5$: \fillin{negative}\\ $x=4$: \fillin{DNE (corner)}\\ $x=7$: \fillin{DNE (endpoint)}
\end{minipage}
\end{nrow}
\begin{nrow}{The Mean Value Theorem (MVT)}
\sg{xmin=-0.3,xmax=3.4,ymin=-0.5,ymax=3.5,width=8cm,height=4.8cm,xtick={0,1.732,3},xticklabels={$a$,$c$,$b$},ytick=\empty}{\addplot[apblue,very thick,domain=0:3.2,samples=60]{x^3/9};\pt{0,0}\pt{3,3}\ifkey\addplot[apred,thick] coordinates {(0,0)(3,3)};\addplot[apgreen!70!black,thick,domain=0.8:2.7]{0.5774+(x-1.732)};\pt{1.732,0.5774}\node[apred,font=\scriptsize] at (axis cs:2.3,2.7){secant};\node[apgreen!70!black,font=\scriptsize] at (axis cs:2.7,1.2){tangent};\fi}
\begin{tabular}{|l|p{0.74\linewidth}|}\hline
\textbf{Needs} & If $f$ is \fillin{continuous} on $[a,b]$ and \fillin{differentiable} on $(a,b)$,\\\hline
\textbf{Proves} & then there is a $c$ in $(a,b)$ with \fillin{$f'(c)=\dfrac{f(b)-f(a)}{b-a}$}.\\\hline
\end{tabular}\par\smallskip
In words: the slope of the \fillin{tangent} at some point equals the slope of the \fillin{secant} (the average rate of change). The MVT is about \fillin{slopes}.
\end{nrow}
\begin{nrow}{Example 1}
$f$ is a differentiable function with $f(2)=5$ and $f(7)=20$. Explain why there must be a value $c$, $2<c<7$, with $f'(c)=3$.
\begin{workspace}[2.4cm]
$f$ is differentiable, so it is also continuous on $[2,7]$: the MVT applies. $\dfrac{f(7)-f(2)}{7-2}=\dfrac{20-5}{5}=3$, so there is a $c$ in $(2,7)$ with $f'(c)=3$.
\end{workspace}
\end{nrow}
\begin{nrow}{Example 2}
\begin{minipage}[t]{0.45\linewidth}\vspace{0pt}
\tinyg{xmin=-1.5,xmax=3.5,ymin=-1.5,ymax=3.5,width=5.4cm,height=4.4cm,xtick={-1,0,1,2,3}}{\addplot[apblue,very thick] coordinates {(-1,3)(1,-1)(3,3)};\pt{-1,3}\pt{3,3}\node[apblue,font=\scriptsize] at (axis cs:2.3,0.6){$g(x)$};}
\end{minipage}\hfill
\begin{minipage}[t]{0.52\linewidth}\vspace{2pt}
$g(x)=2|x-1|-1$ on $[-1,3]$. Find the average rate of change of $g$ on $[-1,3]$. Does the MVT guarantee a $c$ in $(-1,3)$ with $g'(c)$ equal to it?
\end{minipage}
\begin{workspace}[2cm]
$\dfrac{g(3)-g(-1)}{3-(-1)}=\dfrac{3-3}{4}=0$. No: $g$ is not differentiable at $x=1$ (sharp corner), so the MVT does not apply. In fact no point has slope $0$: the slopes are only $-2$ and $2$.
\end{workspace}
\end{nrow}
\begin{nrow}{Summary}
\ifkey{\color{apnavy}MVT: if $f$ is continuous on $[a,b]$ and differentiable on $(a,b)$, some tangent slope equals the average rate of change $\frac{f(b)-f(a)}{b-a}$. Always check differentiability first: one corner is enough to break it.}\else\vspace{2.2cm}\fi
\end{nrow}

%%DOC 1-9 | IVT, EVT, and MVT
\begin{center}\textsf{\small\color{apgrey}Objective: know when to use IVT, EVT and MVT, and justify the answer.}\end{center}
\begin{nrow}{Table review}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{|l|p{0.24\linewidth}|p{0.24\linewidth}|p{0.24\linewidth}|}\hline
 & \centering\textbf{IVT} & \centering\textbf{EVT} & \centering\arraybackslash\textbf{MVT}\\\hline
\textbf{Needs} & \cellans{continuous on $[a,b]$} & \cellans{continuous on a closed interval $[a,b]$} & \cellans{differentiable on $(a,b)$ (so also continuous)}\\\hline
\textbf{Proves} & \cellans{heights (values)} & \cellans{absolute max and min} & \cellans{slopes: $f'(c)=\frac{f(b)-f(a)}{b-a}$}\\\hline
\end{tabular}\par\smallskip
Remember: differentiable $\Rightarrow$ \fillin{continuous}, but continuous does \fillin{not} imply differentiable.
\end{nrow}
\begin{nrow}{Example 1}
$f$ is differentiable for all $x$. \;\begin{tabular}{|c|c|c|c|c|}\hline$x$&$1$&$3$&$5$&$8$\\\hline$f(x)$&$4$&$-2$&$6$&$10$\\\hline\end{tabular}\par
a) Must there be a $c$ in $(1,3)$ with $f(c)=0$? \;b) Must there be a $c$ in $(3,5)$ with $f'(c)=4$? \;c) Must $f$ have an absolute maximum on $[1,8]$?
\begin{workspace}[3cm]
a) Yes, IVT: $f$ is continuous (because differentiable) and $0$ is between $f(1)=4$ and $f(3)=-2$.\\
b) Yes, MVT: $\frac{f(5)-f(3)}{5-3}=\frac{6-(-2)}{2}=4$.\\
c) Yes, EVT: $f$ is continuous on the closed interval $[1,8]$. The maximum is at least $10$.
\end{workspace}
\end{nrow}
\begin{nrow}{Example 2}
$g(x)=\dfrac{1}{x-2}$ on $[1,3]$, with $g(1)=-1$ and $g(3)=1$. Does the IVT guarantee a $c$ with $g(c)=0$?
\begin{workspace}[2cm]
No: $g$ is not continuous on $[1,3]$ (vertical asymptote at $x=2$), so the IVT does not apply. In fact $g(x)=0$ has no solution.
\end{workspace}
\end{nrow}
\begin{nrow}{Example 3}
$h$ is continuous on $[0,4]$, but its graph has a sharp corner at $x=2$. Which theorems apply to $h$ on $[0,4]$?
\begin{workspace}[2cm]
IVT and EVT apply (they only need continuity on $[0,4]$). The MVT does not apply, because $h$ is not differentiable at $x=2$.
\end{workspace}
\end{nrow}
\begin{nrow}{Summary}
\ifkey{\color{apnavy}Name the theorem, check what it needs (continuous, closed interval, or differentiable), then state what it proves: a height (IVT), an absolute max or min (EVT), or a slope (MVT).}\else\vspace{2.2cm}\fi
\end{nrow}
